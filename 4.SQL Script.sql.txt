## 4. SQL Scripts

### 4.1 `db/01-schema.sql` — core (production)

```sql
-- ============================================================
-- File Validation & Ingestion Framework - core schema
-- Requires: PostgreSQL 16+
-- ============================================================
CREATE SCHEMA IF NOT EXISTS fvi;
CREATE EXTENSION IF NOT EXISTS pgcrypto;

-- ---------- users ----------
CREATE TABLE fvi.users (
  user_id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  okta_user_id    VARCHAR(64)  NOT NULL UNIQUE,
  email           VARCHAR(255) NOT NULL UNIQUE,
  display_name    VARCHAR(200) NOT NULL,
  region_code     VARCHAR(8),
  roles           TEXT[] NOT NULL DEFAULT '{}',
  access_status   VARCHAR(16) NOT NULL DEFAULT 'pending'
                  CHECK (access_status IN ('pending','approved','revoked')),
  snow_request_id VARCHAR(40),
  is_active       BOOLEAN NOT NULL DEFAULT TRUE,
  last_login_at   TIMESTAMPTZ,
  last_logout_at  TIMESTAMPTZ,
  session_epoch   INT NOT NULL DEFAULT 1,
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- ---------- reference_data ----------
-- One typed-JSON table for regions, schema types, schema fields,
-- volume mappings, validation rules.
CREATE TABLE fvi.reference_data (
  ref_id      UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  category    VARCHAR(32) NOT NULL
              CHECK (category IN ('region','schema_type','schema_field',
                                  'volume_mapping','validation_rule')),
  key         VARCHAR(128) NOT NULL,
  parent_key  VARCHAR(128),
  position    INT,
  value       JSONB NOT NULL,
  is_active   BOOLEAN NOT NULL DEFAULT TRUE,
  created_at  TIMESTAMPTZ NOT NULL DEFAULT now(),
  UNIQUE (category, key, parent_key)
);
CREATE INDEX ix_ref_cat_parent
  ON fvi.reference_data(category, parent_key) WHERE is_active;

-- ---------- upload_batches ----------
CREATE TABLE fvi.upload_batches (
  batch_id        UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  correlation_id  UUID NOT NULL UNIQUE,
  user_id         UUID NOT NULL REFERENCES fvi.users(user_id),
  schema_type_key VARCHAR(64) NOT NULL,
  region_code     VARCHAR(8)  NOT NULL,
  total_files     INT NOT NULL CHECK (total_files > 0),
  status          VARCHAR(16) NOT NULL DEFAULT 'INITIATED'
                  CHECK (status IN ('INITIATED','PROCESSING','PARTIAL_SUCCESS','COMPLETED','FAILED')),
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  completed_at    TIMESTAMPTZ
);

-- ---------- files ----------
CREATE TABLE fvi.files (
  file_id           UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  batch_id          UUID REFERENCES fvi.upload_batches(batch_id),
  sha256_hash       CHAR(64) NOT NULL,
  file_name         VARCHAR(512) NOT NULL,
  file_size_bytes   BIGINT NOT NULL,
  file_extension    VARCHAR(8)  NOT NULL CHECK (file_extension IN ('csv','xlsx')),
  region_code       VARCHAR(8)  NOT NULL,
  schema_type_key   VARCHAR(64) NOT NULL,
  uploaded_by       UUID REFERENCES fvi.users(user_id),
  machine_host      VARCHAR(255),
  ip_address        INET,
  user_agent        TEXT,
  source_id         UUID,  -- FK added in db/07-channel-schema.sql
  source_ref        VARCHAR(1024),
  status            VARCHAR(32) NOT NULL DEFAULT 'HASH_VALIDATION'
                    CHECK (status IN ('QUEUED','HASH_VALIDATION','DUPLICATE_DETECTED',
                                      'VALIDATION_FAILED','APPROVED',
                                      'TRANSFERRED_TO_DATABRICKS',
                                      'PROCESSING_COMPLETED','PROCESSING_FAILED')),
  reject_reason     TEXT,
  destination_path  VARCHAR(1024),
  status_updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  created_at        TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE UNIQUE INDEX ux_files_sha256 ON fvi.files (sha256_hash);
CREATE INDEX ix_files_batch ON fvi.files(batch_id);
CREATE INDEX ix_files_status_queued ON fvi.files(status)
  WHERE status IN ('QUEUED','HASH_VALIDATION','APPROVED');
CREATE INDEX ix_files_user ON fvi.files(uploaded_by, created_at DESC);

-- ---------- file_events ----------
CREATE TABLE fvi.file_events (
  event_id       BIGSERIAL PRIMARY KEY,
  correlation_id UUID,
  file_id        UUID REFERENCES fvi.files(file_id),
  batch_id       UUID REFERENCES fvi.upload_batches(batch_id),
  event_type     VARCHAR(32) NOT NULL,
  actor_type     VARCHAR(8) NOT NULL DEFAULT 'USER' CHECK (actor_type IN ('USER','SYSTEM')),
  actor_user_id  UUID REFERENCES fvi.users(user_id),
  payload        JSONB NOT NULL DEFAULT '{}',
  ip_address     INET,
  machine_host   VARCHAR(255),
  occurred_at    TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX ix_fe_file   ON fvi.file_events(file_id, occurred_at DESC);
CREATE INDEX ix_fe_batch  ON fvi.file_events(batch_id, occurred_at DESC);
CREATE INDEX ix_fe_corr   ON fvi.file_events(correlation_id);
CREATE INDEX ix_fe_type   ON fvi.file_events(event_type, occurred_at DESC);

-- ---------- transfers ----------
CREATE TABLE fvi.transfers (
  transfer_id     UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  file_id         UUID NOT NULL REFERENCES fvi.files(file_id) ON DELETE CASCADE,
  attempt_number  INT  NOT NULL DEFAULT 1,
  destination     VARCHAR(1024) NOT NULL,
  status          VARCHAR(16) NOT NULL CHECK (status IN ('SUCCESS','FAILED','RETRY')),
  acknowledgement JSONB,
  error_message   TEXT,
  next_retry_at   TIMESTAMPTZ,
  attempted_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
  UNIQUE (file_id, attempt_number)
);

-- ---------- reconciliation ----------
CREATE TABLE fvi.reconciliation (
  recon_id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  file_id           UUID NOT NULL REFERENCES fvi.files(file_id),
  app_status        VARCHAR(32) NOT NULL,
  databricks_status VARCHAR(24),
  databricks_run_id VARCHAR(128),
  records_ingested  BIGINT,
  is_consistent     BOOLEAN NOT NULL,
  details           JSONB,
  reconciled_at     TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX ix_recon_bad ON fvi.reconciliation(reconciled_at DESC) WHERE is_consistent = FALSE;

-- ---------- app_config ----------
CREATE TABLE fvi.app_config (
  config_key   VARCHAR(64) PRIMARY KEY,
  config_value JSONB NOT NULL,
  description  TEXT
);

-- ---------- sso_sessions ----------
CREATE TABLE fvi.sso_sessions (
  session_id      UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id         UUID NOT NULL REFERENCES fvi.users(user_id),
  okta_session_id VARCHAR(128) NOT NULL,
  issued_at       TIMESTAMPTZ NOT NULL,
  expires_at      TIMESTAMPTZ NOT NULL,
  revoked_at      TIMESTAMPTZ,
  revoked_reason  VARCHAR(64),
  ip_address      INET,
  user_agent      TEXT,
  session_epoch   INT NOT NULL,
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX ix_sso_sessions_user   ON fvi.sso_sessions(user_id, issued_at DESC
